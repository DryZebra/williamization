def tts_to_wav(text_path: Path, out_wav: Path, engine: str):
    """Very minimal TTS hook. Choose one:
       - 'none'  : no VO is produced (music-only video)
       - 'pyttsx3': local/offline (pip install pyttsx3; voices vary by OS)
       - 'eleven' : fill in your API call here (placeholder)
    """
    if engine == "none":
        return None
    if engine == "pyttsx3":
        import pyttsx3
        engine_tts = pyttsx3.init()
        with open(text_path, "r", encoding="utf-8") as f:
            txt = f.read()
        engine_tts.save_to_file(txt, str(out_wav))
        engine_tts.runAndWait()
        return out_wav
    if engine == "eleven":
        raise NotImplementedError("Wire up your ElevenLabs/OpenAI TTS here.")
    raise ValueError("Unknown TTS engine")

    rng = np.random.default_rng(seed)
    # base dark teal
    base = np.zeros((h, w, 3), dtype=np.float32)
    base[:,:,1] = 0.12

    cx, cy = w/2, h*0.35

    # normalized coords
