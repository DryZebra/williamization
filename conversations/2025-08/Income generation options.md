import re, textwrap
import feedparser
from bs4 import BeautifulSoup

FEED = "https://vestnikofthelight.substack.com/feed"
INDEX = "index.html"

def get_latest():
    feed = feedparser.parse(FEED)
    entry = feed.entries[0]
    title = entry.title
    url = entry.link
    # extract a short excerpt from content/summary
    html = entry.get("content",[{"value":entry.get("summary","")}])[0]["value"]
    soup = BeautifulSoup(html, "html.parser")
    text = " ".join(soup.get_text().split())
    excerpt = text[:220] + ("…" if len(text) > 220 else "")
    return title, url, excerpt

title, url, excerpt = get_latest()

with open(INDEX, "r", encoding="utf-8") as f:
    html = f.read()

html = re.sub(r"<!--SERMON_TITLE-->.*?<!--/SERMON_TITLE-->",

html = re.sub(r"<!--SERMON_URL-->.*?<!--/SERMON_URL-->",

html = re.sub(r"<!--SERMON_EXCERPT-->.*?<!--/SERMON_EXCERPT-->",
