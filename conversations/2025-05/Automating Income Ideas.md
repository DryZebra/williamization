import os
import time
import pyautogui
import pyperclip

# Config
script_dir = "C:\\FacelessMachine\\scripts\\2025-05-27\\"
delay_after_paste = 3
wait_for_response = 15

# Get all script files
files = [f for f in os.listdir(script_dir) if f.endswith(".txt")]

for filename in files:
    full_path = os.path.join(script_dir, filename)
    
    with open(full_path, "r", encoding="utf-8") as f:
        content = f.read()

    if "[INSERT RESPONSE HERE]" not in content:
        continue

    # Extract prompt
    start = content.find("# Prompt:\n") + len("# Prompt:\n")
    end = content.find("\n\n# William's Response:")
    prompt = content[start:end].strip()

    # Copy to clipboard
    pyperclip.copy(prompt)
    time.sleep(1)

    # Activate ChatGPT window (assumes it's already open and top-left)
    pyautogui.click(x=100, y=100)  # Adjust if needed
    time.sleep(1)

    # Paste + submit
    pyautogui.hotkey("ctrl", "v")
    time.sleep(0.5)
    pyautogui.press("enter")
    time.sleep(wait_for_response)

    # Copy result from screen
    pyautogui.hotkey("ctrl", "a")
    time.sleep(0.5)
    pyautogui.hotkey("ctrl", "c")
    time.sleep(1)

    response = pyperclip.paste()

    # Replace placeholder in original file
    new_content = content.replace("[INSERT RESPONSE HERE]", response)
    with open(full_path, "w", encoding="utf-8") as f:
        f.write(new_content)

    print(f"Filled {filename}")
    time.sleep(2)
