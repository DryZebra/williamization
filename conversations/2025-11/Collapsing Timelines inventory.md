```python
import os, time, datetime, git

REPO_PATH = r"C:\Projects\Vestnik_Cycle"
INTERVAL_MIN = 15  # autosave every 15 minutes

def commit_all(repo):
    repo.git.add(A=True)
    stamp = datetime.datetime.now().strftime("%Y-%m-%d_%H-%M")
    repo.index.commit(f"Auto-save {stamp}")

if __name__ == "__main__":
    repo = git.Repo(REPO_PATH)
    while True:
        commit_all(repo)
        print(f"[{datetime.datetime.now()}] Autosaved.")
        time.sleep(INTERVAL_MIN * 60)
```

Then run this script minimized; it will save every 15 min without you touching anything.

## 🔍 5. Contradiction & Canon Checker (Basic Version)

This script parses your Obsidian Vault for repeated entities (names, dates, traits) and warns you of inconsistencies.

```python
import re, os, json, difflib

VAULT = r"C:\Projects\Vestnik_Cycle\01_Story"
CANON = r"C:\Projects\Vestnik_Cycle\02_Canon\canon_state.json"

def load_texts():
    for root, _, files in os.walk(VAULT):
        for f in files:
            if f.endswith(".md"):
                with open(os.path.join(root, f), encoding="utf8") as fh:
                    yield fh.read()

def find_terms(texts):
    names = re.findall(r"\b[A-Z][a-z]{2,}\b", " ".join(texts))
    return {n for n in names if len(n) > 3}

def detect_conflicts(canon, current):
    return [n for n in current if n not in canon["characters"]]

if __name__ == "__main__":
    texts = list(load_texts())
    current_terms = find_terms(texts)
    canon = json.load(open(CANON, encoding="utf8"))
    conflicts = detect_conflicts(canon, current_terms)
    if conflicts:
        print("⚠️ New or conflicting entities detected:", conflicts)
    else:
        print("✅ Canon consistent.")
```
