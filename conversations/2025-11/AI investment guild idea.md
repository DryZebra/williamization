  import json

          [f for f in os.listdir(LOG_PATH) if f.endswith(".json")],
          reverse=True
      )[:n_files]

      lines = []
      for fname in files:
          path = os.path.join(LOG_PATH, fname)
          with open(path) as f:
              data = json.load(f)

          prices = []
          for ticker, info in data.items():
              # Try to grab ask price if present
              quote = info.get("quote") or {}
              ap = quote.get("ap")
              if isinstance(ap, (int, float)):
                  prices.append(ap)

          if prices:
              avg_price = statistics.mean(prices)
              lines.append(f"{fname}: avg ask across {len(prices)} symbols = ${avg_price:.2f}")
          else:
              lines.append(f"{fname}: no valid prices found")

      if not lines:
          return "No log files found."
      return "\n".join(lines)

  if __name__ == "__main__":
      print(summarize_recent())

from dotenv import load_dotenv

load_dotenv()

SECRET_KEY = os.getenv("APCA_API_SECRET_KEY")

def fetch_account():

def save_snapshot(data: dict) -> str:
    """Save snapshot JSON into the Paper_Trading_Logs folder."""
    os.makedirs(LOG_DIR, exist_ok=True)
    timestamp = datetime.now().strftime("%Y-%m-%d_%H-%M-%S")
    filename = f"account_snapshot_{timestamp}.json"
    path = os.path.join(LOG_DIR, filename)

    with open(path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)

    return path


def main():
    try:
        account = fetch_account()
        path = save_snapshot(account)
        print("✅ Snapshot saved to:", path)
        print("   Equity:", account.get("equity"))
        print("   Cash:", account.get("cash"))
    except Exception as e:

if __name__ == "__main__":

import os
import json
from datetime import datetime

LOG_DIR = os.path.join("04_TESTING", "Paper_Trading_Logs")


def find_latest_snapshot() -> str | None:
    """Return path to most recent account_snapshot_*.json file, or None."""
    if not os.path.isdir(LOG_DIR):
        return None

    files = [
        f for f in os.listdir(LOG_DIR)
        if f.startswith("account_snapshot_") and f.endswith(".json")
    ]
    if not files:
        return None

    files.sort()  # filenames contain timestamp, so lexicographic sort works
    latest = files[-1]
    return os.path.join(LOG_DIR, latest)


def load_snapshot(path: str) -> dict:
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)


def summarize(account: dict) -> str:
