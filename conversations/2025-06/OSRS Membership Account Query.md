function refreshXP(playerCell, targetSheet) {

  const url = "https://secure.runescape.com/m=hiscore_oldschool_ironman/index_lite.ws?player=" + encodeURIComponent(name);
  const csv = UrlFetchApp.fetch(url).getContentText();
  const data = Utilities.parseCsv(csv);

  for (let i = 0; i < skills.length; i++) {
